-- =========================
-- MEDIA (GLOBAL)
-- =========================
CREATE TABLE media (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    url TEXT NOT NULL,
    type ENUM('IMAGE','VIDEO','DOCUMENT') NOT NULL,
    processing_status ENUM('UPLOADING','PROCESSING','READY','FAILED') DEFAULT 'READY',
    file_size INT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_type (type),
    INDEX idx_status (processing_status)
);

-- =========================
-- USERS & AUTH
-- =========================
CREATE TABLE users (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    email VARCHAR(150) UNIQUE NOT NULL,
    username VARCHAR(50) UNIQUE NOT NULL,
    encrypted_name VARCHAR(255) UNIQUE NOT NULL,
    password_hash VARCHAR(255),
    is_verified BOOLEAN DEFAULT FALSE,
    status ENUM('ACTIVE','SUSPENDED','BANNED') DEFAULT 'ACTIVE',
    last_login_at TIMESTAMP NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_email (email),
    INDEX idx_username (username),
    INDEX idx_status (status),
    INDEX idx_created_at (created_at DESC)
);

-- USER SESSIONS
CREATE TABLE user_sessions (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    user_id BIGINT NOT NULL,
    token TEXT NOT NULL,
    refresh_token TEXT,
    expires_at TIMESTAMP NOT NULL,
    refresh_expires_at TIMESTAMP,
    is_active BOOLEAN DEFAULT TRUE,
    ip_address VARCHAR(50),
    user_agent TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    INDEX idx_user_id (user_id),
    INDEX idx_expires_at (expires_at),
    INDEX idx_is_active (is_active)
);

-- OTP (ONE-TIME PASSWORD)
CREATE TABLE otps (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    user_id BIGINT NOT NULL,
    email VARCHAR(150) NOT NULL,
    otp_code VARCHAR(6) NOT NULL,
    purpose ENUM('LOGIN','PASSWORD_RESET','EMAIL_VERIFICATION') NOT NULL,
    is_used BOOLEAN DEFAULT FALSE,
    expires_at TIMESTAMP NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    INDEX idx_user_id (user_id),
    INDEX idx_email (email),
    INDEX idx_expires_at (expires_at),
    INDEX idx_purpose (purpose)
);

-- PASSWORD HISTORY (FOR TRACKING PASSWORD CHANGES)
CREATE TABLE password_history (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    user_id BIGINT NOT NULL,
    old_password_hash VARCHAR(255) NOT NULL,
    changed_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    INDEX idx_user_id (user_id)
);

-- =========================
-- USER PROFILES
-- =========================

CREATE TABLE user_profiles (

    avatar_media_id BIGINT NULL,

    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

-- FOLLOW SYSTEM
CREATE TABLE follows (

-- =========================
-- POSTS
-- =========================
CREATE TABLE posts (

-- =========================
-- TRIGGERS (MAINTENANCE)
-- =========================
DELIMITER //

CREATE TRIGGER update_post_stats_on_reaction
AFTER INSERT ON reactions
FOR EACH ROW
BEGIN
    IF NEW.target_type = 'POST' THEN
        INSERT INTO post_stats (post_id, likes_count) 
        VALUES (NEW.target_id, 1)
        ON DUPLICATE KEY UPDATE 
        likes_count = likes_count + 1,
        last_updated = CURRENT_TIMESTAMP;
    END IF;
END //

CREATE TRIGGER update_post_stats_on_comment
AFTER INSERT ON post_comments
FOR EACH ROW
BEGIN
    INSERT INTO post_stats (post_id, comments_count) 
    VALUES (NEW.post_id, 1)
    ON DUPLICATE KEY UPDATE 
    comments_count = comments_count + 1,
    last_updated = CURRENT_TIMESTAMP;
END //

CREATE TRIGGER update_post_stats_on_rumor_vote
AFTER INSERT ON rumor_votes
FOR EACH ROW
BEGIN
    IF NEW.vote = 'TRUE' THEN
        INSERT INTO post_stats (post_id, rumor_true_votes)
        VALUES (NEW.post_id, 1)
        ON DUPLICATE KEY UPDATE 
        rumor_true_votes = rumor_true_votes + 1,
        last_updated = CURRENT_TIMESTAMP;
    ELSE
        INSERT INTO post_stats (post_id, rumor_false_votes)
        VALUES (NEW.post_id, 1)
        ON DUPLICATE KEY UPDATE 
        rumor_false_votes = rumor_false_votes + 1,
        last_updated = CURRENT_TIMESTAMP;
    END IF;
END //

DELIMITER ;
